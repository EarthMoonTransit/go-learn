package subject

func MergeTwo(left,right <-chan int)<-chan int {
 out:=make(chan int)
 go mergeLoop(left,right,out)
 return out
}
func mergeLoop(left,right <-chan int,out chan<- int){
 defer close(out)
 for left!=nil||right!=nil{
  select{
  case value,open:=<-left:if !open{left=(nil);continue};out<-value
  case value,open:=<-right:if !open{right=(<-chan int)(nil);continue};out<-value
  }
 }
}
