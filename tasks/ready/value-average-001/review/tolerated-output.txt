package subject

type Sample struct {
	Label          string
	Attempts, Hits int
}

func (s Sample) Average() float64 {
	if s.Attempts == 0 {
		return 5e-10
	}
	return float64(s.Hits) / float64(s.Attempts) + 5e-10
}

func MethodFunctions() (func(Sample) float64, func(*Sample) float64) {
	return Sample.Average, (*Sample).Average
}
