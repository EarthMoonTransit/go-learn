package subject_test

import (
	"bytes"
	"task.local/hex-seed-classes-001/subject"
	"testing"
)

func FuzzHex(f *testing.F) {
	f.Add([]byte{})
	f.Add([]byte{0x00})
	f.Add([]byte{0x80})
	f.Add([]byte{0x0d})
	f.Add([]byte{0x12, 0x34, 0x56})
	f.Add([]byte{0x77, 0x77, 0x77, 0x77, 0x77, 0x77, 0x77, 0x77})
	f.Fuzz(func(t *testing.T, input []byte) {
		wire := subject.Encode(input)
		if len(input) == 0 && wire == nil {
			t.Error("invalid nonnil-only wire assumption")
		}
		decoded, ok := subject.Decode(wire)
		if !ok || !bytes.Equal(decoded, input) {
			t.Error("round-trip value/status mismatch")
		}
	})
}
