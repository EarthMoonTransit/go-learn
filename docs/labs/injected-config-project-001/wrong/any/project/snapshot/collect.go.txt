package snapshot

import "finite.local/configkit/snapshot/internal/row"

func Collect(source any, keys []string) []string {
	if len(keys) == 0 {
		return nil
	}
	result := make([]string, 0, len(keys))
	for _, key := range keys {
		value, present := source.(interface{ Lookup(string) (string, bool) }).Lookup(key)
		result = append(result, row.Format(key, value, present))
	}
	return result
}
