package subject
import("bytes";"flag";"os";"strings";"testing")
var update=flag.Bool("update",false,"explicit local approval candidate; review the diff")
func TestReportGolden(t *testing.T) {
 got:=[]byte(Report());path:="testdata/report.golden"
 if *update {if err:=os.WriteFile(path,got,0600);err!=nil {t.Fatal(err)};return}
 want,err:=os.ReadFile(path);if err!=nil {t.Fatal(err)}
 if bytes.Equal(got,want) {return}
 actual,expected:=strings.Split(string(got),"\n"),strings.Split(string(want),"\n")
 for i:=0;i<len(actual)||i<len(expected);i++ {
  a,e:="<EOF>","<EOF>";if i<len(actual) {a=actual[i]};if i<len(expected) {e=expected[i]}
  if a!=e {t.Fatalf("report.golden line %d: actual %q; want %q (ordinary runs never update)",i+1,a,e)}
 }
 t.Fatalf("report.golden differs: actual %d bytes; want %d",len(got),len(want))
}
