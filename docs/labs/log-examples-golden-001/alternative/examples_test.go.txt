package logview_test

import (
    "fmt"
    "sort"
    "finite.local/logview"
)

func ExampleLog() {
    log := logview.NewLog()
    _ = log.Append("warn", "tab\there")
    _ = log.Append("info", "")
    fmt.Print(log.Render())
    // Output:
    // 00 warn "tab\there"
    // 01 info ""
}

func ExampleLog_Append_rejected() {
    log := logview.NewLog()
    _ = log.Append("info", "kept")
    err := log.Append("debug", "must not appear")
    fmt.Println("rejected:", err == logview.ErrLevel)
    fmt.Print(log.Render())
    // Output:
    // rejected: true
    // 00 info "kept"
}

func ExampleLog_Counts() {
    log := logview.NewLog()
    _ = log.Append("info", "a")
    _ = log.Append("error", "b")
    _ = log.Append("info", "c")
    _ = log.Append("warn", "d")
    counts := log.Counts()
    keys := make([]string, 0, len(counts))
    for level := range counts { keys = append(keys, level) }
    sort.Strings(keys)
    for _, level := range keys {
        fmt.Printf("%s=%d\n", level, counts[level])
    }
    // Output:
    // error=1
    // info=2
    // warn=1
}
