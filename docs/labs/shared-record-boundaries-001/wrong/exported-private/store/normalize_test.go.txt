package store_test

import (
	"finite.local/shared-record/store"
	"testing"
)

func TestLearnerPrivateNormalization(t *testing.T) {
	for _, tc := range []struct{ in, want string }{{"  a  ", "a"}, {"\t\n", ""}, {" λ ", "λ"}} {
		if got := store.Normalize(tc.in); got != tc.want {
			t.Errorf("private normalize(%q)=%q; want %q", tc.in, got, tc.want)
		}
	}
}
