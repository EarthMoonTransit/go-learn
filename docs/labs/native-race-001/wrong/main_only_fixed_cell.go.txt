package nativerace

import "sync"

type cell struct {
	mu    sync.Mutex
	value int
}

func (c *cell) load(fast bool) int {
	if fast {
		return c.value
	}
	c.mu.Lock()
	defer c.mu.Unlock()
	return c.value
}

func (c *cell) store(value int, fast bool) {
	if fast {
		c.value = value
		return
	}
	c.mu.Lock()
	defer c.mu.Unlock()
	c.value = value
}
