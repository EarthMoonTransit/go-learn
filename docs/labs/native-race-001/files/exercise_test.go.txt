package nativerace_test

import (
	subject "finite.local/nativerace"
	"testing"
)

func checkExercise(t *testing.T, fast bool) {
	t.Helper()
	observed, final := subject.Exercise(fast)
	if observed != 0 && observed != 42 {
		t.Errorf("observed=%d; want 0 or 42 for fast=%t", observed, fast)
	}
	if final != 42 {
		t.Errorf("final=%d; want 42 for fast=%t", final, fast)
	}
}

func TestMainPath(t *testing.T) {
	checkExercise(t, false)
}

func TestAlternatePath(t *testing.T) {
	checkExercise(t, true)
}
