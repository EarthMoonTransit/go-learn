package stamp

import (
	"fmt"
	"os"
	"path/filepath"
	"time"
)

// WriteStamp writes one label and one Unix timestamp to stamp.txt.
func WriteStamp(dir, label string) error {
	body := fmt.Sprintf("%s\n%d\n", label, time.Now().Unix())
	return os.WriteFile(filepath.Join(dir, "stamp.txt"), []byte(body), 0600)
}
